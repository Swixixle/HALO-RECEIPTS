## 1) Task 1: `auditPayloadV1()` with `_v: 1` marker inside the hashed payload

### `auditPayload.ts`

```ts
// auditPayload.ts
export type AuditRow = {
  seq: number;
  ts: string;              // ISO string or DB timestamp rendered consistently
  actor: string | null;
  action: string;          // event type
  entityType: string | null;
  entityId: string | null;
  requestId: string | null;

  // canonical event data (MUST already be JSON-safe)
  data: unknown;

  // chain linkage
  prevHash: string;        // hex string
};

export type AuditPayloadV1 = {
  _v: 1;
  seq: number;
  ts: string;
  actor: string | null;
  action: string;
  entityType: string | null;
  entityId: string | null;
  requestId: string | null;
  prevHash: string;
  data: unknown;
};

/**
 * Canonical, hashed payload for v1.
 * IMPORTANT: append() and verify() must both call this.
 */
export function auditPayloadV1(row: AuditRow): AuditPayloadV1 {
  return {
    _v: 1,
    seq: row.seq,
    ts: row.ts,
    actor: row.actor ?? null,
    action: row.action,
    entityType: row.entityType ?? null,
    entityId: row.entityId ?? null,
    requestId: row.requestId ?? null,
    prevHash: row.prevHash,
    data: row.data,
  };
}
```

**Key point:** `_v: 1` is inside the hashed bytes, so the hash is *self-describing*.

## 2) Task 2: `stableStringifyStrict()` rejects dangerous/ambiguous types

### `stableStringifyStrict.ts`

```ts
// stableStringifyStrict.ts
function isPlainObject(x: unknown): x is Record<string, unknown> {
  if (x === null || typeof x !== "object") return false;
  const proto = Object.getPrototypeOf(x);
  return proto === Object.prototype || proto === null;
}

function fail(path: string, msg: string): never {
  throw new Error(`stableStringifyStrict: ${msg} at ${path}`);
}

/**
 * Deterministic JSON serialization with strict type rejection.
 * - Sorts object keys (lexicographic)
 * - Rejects undefined, BigInt, Date, NaN/Infinity, functions, symbols, class instances, Map/Set, etc.
 * - Rejects circular structures
 */
export function stableStringifyStrict(value: unknown): string {
  const seen = new WeakSet<object>();

  const walk = (v: unknown, path: string): unknown => {
    // null
    if (v === null) return null;

    const t = typeof v;

    if (t === "string" || t === "boolean") return v;

    if (t === "number") {
      if (!Number.isFinite(v)) fail(path, "non-finite number (NaN/Infinity)");
      return v;
    }

    if (t === "undefined") fail(path, "undefined is not allowed");
    if (t === "bigint") fail(path, "BigInt is not allowed");
    if (t === "function") fail(path, "function is not allowed");
    if (t === "symbol") fail(path, "symbol is not allowed");

    // objects
    if (t === "object") {
      // Date (and other class instances)
      if (v instanceof Date) fail(path, "Date is not allowed");
      if (v instanceof Map) fail(path, "Map is not allowed");
      if (v instanceof Set) fail(path, "Set is not allowed");
      if (v instanceof RegExp) fail(path, "RegExp is not allowed");
      if (typeof Buffer !== "undefined" && Buffer.isBuffer(v)) fail(path, "Buffer is not allowed");

      // arrays
      if (Array.isArray(v)) {
        if (seen.has(v)) fail(path, "circular reference");
        seen.add(v);
        return v.map((item, i) => walk(item, `${path}[${i}]`));
      }

      // plain objects only
      if (!isPlainObject(v)) fail(path, "non-plain object (class/instance) is not allowed");

      if (seen.has(v)) fail(path, "circular reference");
      seen.add(v);

      const obj = v as Record<string, unknown>;
      const keys = Object.keys(obj).sort();
      const out: Record<string, unknown> = {};
      for (const k of keys) {
        if (k === "__proto__" || k === "constructor" || k === "prototype") {
          fail(`${path}.${k}`, "dangerous key is not allowed");
        }
        out[k] = walk(obj[k], `${path}.${k}`);
      }
      return out;
    }

    // should be unreachable
    fail(path, `unsupported type: ${t}`);
  };

  const normalized = walk(value, "$");
  return JSON.stringify(normalized);
}
```

This is deliberately “cryptographically boring.” If a caller tries to log weird types, they get a **hard error** *before* hashing.

## 3) Hashing: one function, called by append + verify

### `auditHash.ts`

```ts
// auditHash.ts
import { createHash } from "crypto";
import { stableStringifyStrict } from "./stableStringifyStrict";

export function sha256Hex(input: string): string {
  return createHash("sha256").update(input, "utf8").digest("hex");
}

export function hashAuditPayload(payload: unknown): string {
  return sha256Hex(stableStringifyStrict(payload));
}
```

## 4) Append uses `auditPayloadV1()` (no drift)

Pseudo-structure below assumes you have `audit_events` table + `audit_head` row, and you append inside a transaction with a row lock.

### `appendAuditEvent()`

```ts
import { auditPayloadV1 } from "./auditPayload";
import { hashAuditPayload } from "./auditHash";

export async function appendAuditEvent(tx: any, input: {
  ts: string;
  actor: string | null;
  action: string;
  entityType: string | null;
  entityId: string | null;
  requestId: string | null;
  data: unknown;
}) {
  // 1) lock head row (FOR UPDATE)
  const head = await tx.getAuditHeadForUpdate(); // must lock
  const nextSeq = head.nextSeq;                 // e.g. 1-based
  const prevHash = head.headHash;               // "0".repeat(64) for genesis

  // 2) build canonical row and payload
  const row = {
    seq: nextSeq,
    ts: input.ts,
    actor: input.actor ?? null,
    action: input.action,
    entityType: input.entityType ?? null,
    entityId: input.entityId ?? null,
    requestId: input.requestId ?? null,
    data: input.data,
    prevHash,
  };

  const payload = auditPayloadV1(row);
  const hash = hashAuditPayload(payload);

  // 3) insert event row
  await tx.insertAuditEvent({
    seq: row.seq,
    ts: row.ts,
    actor: row.actor,
    action: row.action,
    entity_type: row.entityType,
    entity_id: row.entityId,
    request_id: row.requestId,
    prev_hash: row.prevHash,
    hash,
    data: row.data,
    payload_v: 1, // optional but recommended for Path A
  });

  // 4) advance head
  await tx.updateAuditHead({
    headSeq: row.seq,
    headHash: hash,
    nextSeq: row.seq + 1,
  });

  return { seq: row.seq, hash };
}
```

## 5) Verify reports partial coverage + total events

### Result type

```ts
export type AuditVerifyResult =
  | {
      ok: true;
      totalEvents: number;
      verifiedCount: number;
      verifiedFromSeq: number | null;
      verifiedToSeq: number | null;
      partial: boolean;
    }
  | {
      ok: false;
      totalEvents: number;
      verifiedCount: number;
      verifiedFromSeq: number | null;
      verifiedToSeq: number | null;
      partial: boolean;
      reason: "hash_mismatch" | "seq_gap" | "head_mismatch" | "empty";
      atSeq?: number;
      expectedHash?: string;
      actualHash?: string;
      detail?: string;
    };
```

### `verifyAuditChain()`

```ts
import { auditPayloadV1 } from "./auditPayload";
import { hashAuditPayload } from "./auditHash";

export async function verifyAuditChain(db: any, opts: { limit?: number } = {}): Promise<AuditVerifyResult> {
  const totalEvents = await db.countAuditEvents();
  if (totalEvents === 0) {
    return { ok: true, totalEvents: 0, verifiedCount: 0, verifiedFromSeq: null, verifiedToSeq: null, partial: false };
  }

  const limit = opts.limit ?? totalEvents;
  const partial = limit < totalEvents;

  // Get events ordered by seq ASC, limited
  const rows = await db.listAuditEventsAsc({ limit });

  // Basic seq continuity check (within window)
  let verifiedFromSeq: number | null = null;
  let verifiedToSeq: number | null = null;

  for (let i = 0; i < rows.length; i++) {
    const r = rows[i];

    const expectedSeq = rows[0].seq + i;
    if (r.seq !== expectedSeq) {
      return {
        ok: false,
        totalEvents,
        verifiedCount: i,
        verifiedFromSeq: verifiedFromSeq,
        verifiedToSeq: verifiedToSeq,
        partial,
        reason: "seq_gap",
        atSeq: expectedSeq,
        detail: `Expected seq ${expectedSeq} but found ${r.seq}`,
      };
    }

    // Recompute hash from canonical payload
    const payload = auditPayloadV1({
      seq: r.seq,
      ts: r.ts,
      actor: r.actor,
      action: r.action,
      entityType: r.entity_type,
      entityId: r.entity_id,
      requestId: r.request_id,
      data: r.data,
      prevHash: r.prev_hash,
    });

    const computed = hashAuditPayload(payload);

    if (computed !== r.hash) {
      return {
        ok: false,
        totalEvents,
        verifiedCount: i,
        verifiedFromSeq,
        verifiedToSeq,
        partial,
        reason: "hash_mismatch",
        atSeq: r.seq,
        expectedHash: computed,
        actualHash: r.hash,
      };
    }

    if (verifiedFromSeq === null) verifiedFromSeq = r.seq;
    verifiedToSeq = r.seq;
  }

  // Head check ONLY when not partial (otherwise window cannot prove full tail)
  if (!partial) {
    const head = await db.getAuditHead();
    const last = rows[rows.length - 1];
    if (head.head_seq !== last.seq || head.head_hash !== last.hash) {
      return {
        ok: false,
        totalEvents,
        verifiedCount: rows.length,
        verifiedFromSeq,
        verifiedToSeq,
        partial: false,
        reason: "head_mismatch",
        detail: "audit_head does not match last audit_events row",
      };
    }
  }

  return {
    ok: true,
    totalEvents,
    verifiedCount: rows.length,
    verifiedFromSeq,
    verifiedToSeq,
    partial,
  };
}
```

This makes it **impossible** for an operator to confuse “first-N verified” with “whole chain verified.”

## 6) Migration choice

### Path A (recommended): keep old events verifiable (no wipe)

Add a `payload_v` column and implement **both** payload builders:

* `auditPayloadV0()` = your old format (no `_v`)
* `auditPayloadV1()` = new format with `_v: 1`
* Verify uses `row.payload_v` to choose builder
* Append always uses v1 going forward

SQL sketch:

```sql
ALTER TABLE audit_events ADD COLUMN payload_v INT NOT NULL DEFAULT 0;
-- existing rows are v0, new rows become v1
```

### Path B (your described approach): wipe + reset head

If you truly only have 3 events and don’t care about keeping them:

```sql
BEGIN;

DELETE FROM audit_events;

UPDATE audit_head
SET head_seq = 0,
    head_hash = repeat('0', 64),
    next_seq = 1;

COMMIT;
```

If you’re on SQLite, `repeat('0',64)` becomes hardcoded `'0000....'`.

## 7) Frontend: show verified coverage clearly (no overlap / no ambiguity)

Minimal React-ish rendering logic:

```tsx
function AuditVerifyBanner({ r }: { r: AuditVerifyResult }) {
  if (r.ok) {
    if (r.totalEvents === 0) return <div className="banner ok">No audit events.</div>;

    const range =
      r.verifiedFromSeq !== null && r.verifiedToSeq !== null
        ? `Verified seq ${r.verifiedFromSeq}–${r.verifiedToSeq}`
        : "Verified range unknown";

    const suffix = r.partial
      ? ` (window: ${r.verifiedCount}/${r.totalEvents})`
      : ` (${r.verifiedCount}/${r.totalEvents} fully verified)`;

    return <div className="banner ok">{range}{suffix}</div>;
  }

  const where = r.atSeq ? ` at seq ${r.atSeq}` : "";
  const coverage = r.partial ? ` (window: ${r.verifiedCount}/${r.totalEvents})` : ` (${r.verifiedCount}/${r.totalEvents})`;

  return (
    <div className="banner bad">
      Verification failed: {r.reason}{where}{coverage}
    </div>
  );
}
```

If you were seeing “words overlapping,” the fix is usually **CSS constraints** (flex wrap, min-width, and line breaks). Example:

```css
.banner {
  display: flex;
  flex-wrap: wrap;
  gap: 8px;
  padding: 10px 12px;
  border-radius: 10px;
  font-size: 13px;
  line-height: 1.35;
  word-break: break-word;
}
.banner.ok { border: 1px solid rgba(0,0,0,.12); }
.banner.bad { border: 1px solid rgba(255,0,0,.35); }
```

## 8) Threat model “escalation path” note (copy/paste ready)

Add this to your Governance UI / docs:

* **In-DB hash chaining** detects tampering *within the database* (edit/delete/reorder/head mismatch).
* **Escalation path** for stronger adversaries (DB admin compromise): **external anchoring** via:

  * signed checkpoints (operator key),
  * WORM / append-only log sink,
  * or third-party attestation / timestamping.

This signals you understand the difference between **tamper-evident** and **tamper-resistant**.

## 9) The three red-team tests (API-level) you mentioned — canonical expected outcomes

* **Payload edit** → `hash_mismatch` at the tampered `seq`
* **Row delete** → `seq_gap` at expected `seq`
* **Head hash tamper** → `head_mismatch` (or tail `hash_mismatch` depending on your head-check design)

Your verify function above will produce those distinct reasons (with `head_mismatch` only when full verification is run).
